-- ============================================================
-- Ders Programi Uygulamasi - SQLite Schema
-- ============================================================

PRAGMA foreign_keys = ON;

-- ------------------------------------------------------------
-- 1. DEPARTMENT
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS department (
    department_id   TEXT PRIMARY KEY NOT NULL,
    department_name TEXT             NOT NULL
);

-- ------------------------------------------------------------
-- 2. SEMESTER
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS semester (
    semester_id   TEXT    PRIMARY KEY NOT NULL,
    semester_name TEXT                NOT NULL,
    start_date    TEXT                NOT NULL,  -- 'YYYY-MM-DD'
    finish_date   TEXT                NOT NULL,  -- 'YYYY-MM-DD'
    status        INTEGER             NOT NULL DEFAULT 0,
    CONSTRAINT semester_status_check CHECK (status IN (0, 1))
);

-- ------------------------------------------------------------
-- 3. LECTURER
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS lecturer (
    lecturer_id   TEXT PRIMARY KEY NOT NULL,
    department_id TEXT             NOT NULL,
    CONSTRAINT fk_lecturer_department
        FOREIGN KEY (department_id) REFERENCES department(department_id)
);

-- ------------------------------------------------------------
-- 4. USER
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS user (
    user_id     TEXT PRIMARY KEY NOT NULL,
    name_surname TEXT            NOT NULL,
    email        TEXT            NOT NULL UNIQUE,
    password     TEXT            NOT NULL,
    lecturer_id  TEXT,                           -- NULL ise admin
    role         TEXT            NOT NULL,
    CONSTRAINT user_role_check CHECK (role IN ('admin', 'lecturer')),
    CONSTRAINT user_role_lecturer_check CHECK (
        (role = 'lecturer' AND lecturer_id IS NOT NULL) OR
        (role = 'admin'    AND lecturer_id IS NULL)
    ),
    CONSTRAINT fk_user_lecturer
        FOREIGN KEY (lecturer_id) REFERENCES lecturer(lecturer_id)
);

-- ------------------------------------------------------------
-- 5. CLASSROOMS
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS classrooms (
    classroom_id   TEXT    PRIMARY KEY NOT NULL,
    classroom_name TEXT                NOT NULL,
    capacity       INTEGER             NOT NULL,
    lab_existence  INTEGER             NOT NULL DEFAULT 0,
    building       TEXT,
    CONSTRAINT classrooms_lab_check CHECK (lab_existence IN (0, 1))
);

-- ------------------------------------------------------------
-- 6. COURSES
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS courses (
    course_id          TEXT    PRIMARY KEY NOT NULL,
    course_code        TEXT                NOT NULL,
    course_name        TEXT                NOT NULL,
    course_hours       INTEGER             NOT NULL,
    lab_existence      INTEGER             NOT NULL DEFAULT 0,
    lab_hours          INTEGER,                      -- NULL ise lab yok
    lecturer_id        TEXT                NOT NULL,
    course_assistance_id TEXT,                       -- NULL olabilir
    course_capacity    INTEGER             NOT NULL,
    semester_id        TEXT                NOT NULL,
    CONSTRAINT courses_lab_check CHECK (lab_existence IN (0, 1)),
    CONSTRAINT courses_lab_hours_check CHECK (
        (lab_existence = 1 AND lab_hours IS NOT NULL) OR
        (lab_existence = 0 AND lab_hours IS NULL)
    ),
    CONSTRAINT fk_courses_lecturer
        FOREIGN KEY (lecturer_id) REFERENCES lecturer(lecturer_id),
    CONSTRAINT fk_courses_assistance
        FOREIGN KEY (course_assistance_id) REFERENCES lecturer(lecturer_id),
    CONSTRAINT fk_courses_semester
        FOREIGN KEY (semester_id) REFERENCES semester(semester_id)
);

-- ------------------------------------------------------------
-- 7. COURSE_DEPARTMENT  (many-to-many)
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS course_department (
    id            TEXT PRIMARY KEY NOT NULL,
    course_id     TEXT             NOT NULL,
    department_id TEXT             NOT NULL,
    CONSTRAINT fk_cd_course
        FOREIGN KEY (course_id)     REFERENCES courses(course_id),
    CONSTRAINT fk_cd_department
        FOREIGN KEY (department_id) REFERENCES department(department_id),
    CONSTRAINT uq_course_department UNIQUE (course_id, department_id)
);

-- ------------------------------------------------------------
-- 8. AVAILABILITY_STATUS
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS availability_status (
    availability_id TEXT PRIMARY KEY NOT NULL,
    lecturer_id     TEXT             NOT NULL,
    day             TEXT             NOT NULL,
    start_time      TEXT             NOT NULL,  -- 'HH:MM'
    finish_time     TEXT             NOT NULL,  -- 'HH:MM'
    semester_id     TEXT             NOT NULL,
    CONSTRAINT availability_day_check CHECK (
        day IN ('Pazartesi', 'Sali', 'Carsamba', 'Persembe', 'Cuma')
    ),
    CONSTRAINT availability_time_check CHECK (finish_time > start_time),
    CONSTRAINT fk_availability_lecturer
        FOREIGN KEY (lecturer_id)  REFERENCES lecturer(lecturer_id),
    CONSTRAINT fk_availability_semester
        FOREIGN KEY (semester_id)  REFERENCES semester(semester_id)
);

-- ============================================================
-- INDEXLER  (sorgu performansi icin)
-- ============================================================
CREATE INDEX IF NOT EXISTS idx_courses_lecturer    ON courses(lecturer_id);
CREATE INDEX IF NOT EXISTS idx_courses_semester    ON courses(semester_id);
CREATE INDEX IF NOT EXISTS idx_availability_lecturer ON availability_status(lecturer_id);
CREATE INDEX IF NOT EXISTS idx_availability_semester ON availability_status(semester_id);
CREATE INDEX IF NOT EXISTS idx_cd_course           ON course_department(course_id);
CREATE INDEX IF NOT EXISTS idx_cd_department       ON course_department(department_id);
CREATE INDEX IF NOT EXISTS idx_user_email          ON user(email);
